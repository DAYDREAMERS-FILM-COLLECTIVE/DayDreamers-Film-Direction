        <script>
            const burgerToggle = document.getElementById('burgerToggle');
            const sweepWall = document.getElementById('sweepWall');
            const heroView = document.getElementById('heroView');
            const menuView = document.getElementById('menuView');
            const barTop = document.querySelector('.bar-top');
            const barBottom = document.querySelector('.bar-bottom');

            let isOpen = false;
            let isAnimating = false;

            // Optional: Set to true if you want the menu to automatically close 2 seconds after opening
            const AUTO_CLOSE_MENU = false;

            // Start with wall off-screen to the left
            gsap.set(sweepWall, { xPercent: -100 });

            function toggleMenu() {
                if (isAnimating) return; // Strict lock: will not close before wall finishes
                isAnimating = true;

                // 1. Morph burger icon
                if (!isOpen) {
                    gsap.to(barTop, { y: 4, rotate: 45, duration: 0.35, ease: 'power2.inOut' });
                    gsap.to(barBottom, {
                        y: -4,
                        rotate: -45,
                        duration: 0.35,
                        ease: 'power2.inOut',
                    });
                } else {
                    gsap.to([barTop, barBottom], {
                        y: 0,
                        rotate: 0,
                        duration: 0.35,
                        ease: 'power2.inOut',
                    });
                }

                // 2. Continuous Wall Transition with 2-Second Hold at the Right Wall
                const tl = gsap.timeline({
                    onComplete: () => {
                        // Reset wall back to the left off-screen for the next transition
                        gsap.set(sweepWall, { xPercent: -100 });
                        isOpen = !isOpen;
                        isAnimating = false;

                        // If auto-close mode is enabled, trigger closing after menu reveals
                        if (isOpen && AUTO_CLOSE_MENU) {
                            gsap.delayedCall(2.0, () => toggleMenu());
                        }
                    },
                });

                tl
                    // STEP 1: Wall sweeps in from the left until it reaches the right corner (0% = covers viewport)
                    .fromTo(
                        sweepWall,
                        { xPercent: -100 },
                        { xPercent: 0, duration: 0.55, ease: 'power2.inOut' }
                    )

                    // STEP 2: Swap screens behind the wall while screen is 100% blocked
                    .add(() => {
                        if (!isOpen) {
                            heroView.style.display = 'none';
                            menuView.style.display = 'flex';
                        } else {
                            menuView.style.display = 'none';
                            heroView.style.display = 'flex';
                        }
                    })

                    // STEP 3: Hold firmly at the right wall for 2 seconds (does not leave/close before this finishes)
                    .to({}, { duration: 0.5 })

                    // STEP 4: Only after reaching the right wall and waiting 2s, it sweeps off to the right (100%)
                    .to(sweepWall, {
                        xPercent: 100,
                        duration: 0.55,
                        ease: 'power2.inOut',
                    });
            }

            burgerToggle.addEventListener('click', toggleMenu);
        </script>
